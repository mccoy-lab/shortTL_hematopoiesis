Modeling repeated evolution in HSC population

Michael Tassia

March 10, 2026

Theorhetical modeling framework

Load libraries

library(tidyverse)
library(pbapply)
library(patchwork)
library(cowplot)
library(viridis)

Set parameters

Parameter values obtained from empirical estimates in Mitchell et al. 2022. Human genome size is a rough approximation.

hsc_pop <- c(80000,100000,120000) 
ages <- seq(1,80,1)
annual_rates <- 2^seq(4,13) 
diploid_genome_size <- 6.4e9

Run models

Code block will compute \(P(X > 1)\) using Poisson and Binomial distributions (both results are identical). Note that these model the probability of a mutation occurring at the same site in any \(>1\) genomes in the population, disregarding the resulting state of the mutation.

## Poisson model
p.df <- pblapply(hsc_pop, function(x) {
  lapply(annual_rates, function(y) {
    df <- ppois(q = 1,
                lambda = ((y / diploid_genome_size) * x * ages), 
                lower.tail = F) %>% 
      as.data.frame() %>%
      rownames_to_column(var="age") %>%
      mutate(age = as.numeric(age)) %>%
      rename("p" = ".") # set the prob column name to p and convert rowname to age column
    
    df$N <- x # set HSC population size for df
    df$annual_muts <- as.factor(y) # set annual mutation rate for df
    df <- df %>% select(age, N, annual_muts, p)
    return(df)
  })
}) %>% 
  bind_rows() %>% 
  tibble()

## Binomial model
b.df <- pblapply(hsc_pop, function(x) {
  lapply(annual_rates, function(y) {
    df <- pbinom(q = 1,
                 prob = ((y / diploid_genome_size) * ages),
                 size = x,
                 lower.tail = F) %>% 
      as.data.frame() %>%
      rownames_to_column(var = "age") %>%
      mutate(age = as.numeric(age)) %>%
      rename("p" = ".")
    
    df$N <- x
    df$annual_muts <- as.factor(y)
    df<-df %>% select(age, N, annual_muts, p)
    return(df)
  })
}) %>% 
  bind_rows() %>% 
  tibble()
